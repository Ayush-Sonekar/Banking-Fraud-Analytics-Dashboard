-- 18. RULE EFFECTIVENESS

SELECT
    r.Rule_ID,
    r.Rule_Name,
    COUNT(a.Alert_ID) AS alert_count,
    COUNT(DISTINCT a.Transaction_ID) AS transactions_alerted,
    COUNT(DISTINCT i.Investigation_ID) AS investigations,
    COUNT(DISTINCT CASE
        WHEN i.Resolution_Code = 'Confirmed Fraud'
        THEN i.Investigation_ID
    END) AS confirmed_fraud,
    COUNT(DISTINCT CASE
        WHEN i.Resolution_Code = 'False Positive'
        THEN i.Investigation_ID
    END) AS false_positive
FROM fact_alerts a
JOIN dim_rule r
    ON a.Rule_ID = r.Rule_ID
LEFT JOIN fact_investigation i
    ON a.Investigation_ID = i.Investigation_ID
GROUP BY
    r.Rule_ID,
    r.Rule_Name
ORDER BY confirmed_fraud DESC;

30. Fraud by transaction velocity bucket

WITH customer_activity AS (
    SELECT
        Customer_ID,
        COUNT(*) AS transaction_count
    FROM fact_transaction
    GROUP BY Customer_ID
)
SELECT
    CASE
        WHEN transaction_count < 40 THEN 'Low'
        WHEN transaction_count < 80 THEN 'Medium'
        WHEN transaction_count < 120 THEN 'High'

    COUNT(*) AS customers

31. True Transaction Velocity

    t1.Customer_ID,
    t1.Transaction_ID,
    COUNT(t2.Transaction_ID) AS transactions_within_60min
FROM fact_transaction t1
JOIN fact_transaction t2
    ON t1.Customer_ID = t2.Customer_ID
   AND t2.Timestamp BETWEEN t1.Timestamp
                           AND t1.Timestamp + INTERVAL '60 minutes'
GROUP BY
    t1.Customer_ID,
    t1.Transaction_ID
HAVING COUNT(t2.Transaction_ID) >= 3
ORDER BY transactions_within_60min DESC;

32. Fraud by maximum 60-minutes velocity

Transactions
     ⬇️
Calculate each transaction's 60-min window
     ⬇️
Find MAX velocity for each customer
     ⬇️
Assign customer to velocity band
     ⬇️
Measure fraud among that customer's transactions

Query:
WITH velocity AS (
    SELECT
        t1.Customer_ID,
        t1.Transaction_ID,
        t1.Ground_Truth_Label,
        COUNT(t2.Transaction_ID) AS transactions_within_60min
    FROM fact_transaction t1
    JOIN fact_transaction t2
        ON t1.Customer_ID = t2.Customer_ID
       AND t2.Timestamp BETWEEN t1.Timestamp
                            AND t1.Timestamp + INTERVAL '60 minutes'
    GROUP BY
        t1.Customer_ID,
        t1.Transaction_ID,
        t1.Ground_Truth_Label
),

customer_velocity AS (
    SELECT
        Customer_ID,
        MAX(transactions_within_60min) AS max_transactions_60min,
        COUNT(*) AS total_transactions,
        SUM(
            CASE
                WHEN Ground_Truth_Label THEN 1
                ELSE 0
            END
        ) AS fraud_transactions
    FROM velocity
    GROUP BY Customer_ID
),

banded AS (
    SELECT
        CASE
            WHEN max_transactions_60min = 1 THEN '1'
            WHEN max_transactions_60min = 2 THEN '2'
            WHEN max_transactions_60min = 3 THEN '3'
            WHEN max_transactions_60min = 4 THEN '4'
            WHEN max_transactions_60min = 5 THEN '5'
            ELSE '6+'
        END AS velocity_band,
        total_transactions,
        fraud_transactions
    FROM customer_velocity
)

SELECT
    velocity_band,
    COUNT(*) AS customers,
    SUM(total_transactions) AS transactions,
    SUM(fraud_transactions) AS fraud_transactions,
    ROUND(
        100.0 * SUM(fraud_transactions)
        / NULLIF(SUM(total_transactions), 0),
        2
    ) AS fraud_rate_pct
FROM banded
GROUP BY velocity_band
ORDER BY
    CASE velocity_band
        WHEN '1' THEN 1
        WHEN '2' THEN 2
        WHEN '3' THEN 3
        WHEN '4' THEN 4
        WHEN '5' THEN 5
        WHEN '6+' THEN 6
    END;

34. Exact Burst Duration of transaction

WITH velocity AS (
    SELECT
        t1.Customer_ID,
        t1.Transaction_ID,
        t1.Timestamp AS window_start,
        COUNT(t2.Transaction_ID) AS transactions_within_60min,
        MIN(t2.Timestamp) AS first_in_window,
        MAX(t2.Timestamp) AS last_in_window
    FROM fact_transaction t1
    JOIN fact_transaction t2
        ON t1.Customer_ID = t2.Customer_ID
       AND t2.Timestamp BETWEEN t1.Timestamp
                            AND t1.Timestamp + INTERVAL '60 minutes'
    GROUP BY
        t1.Customer_ID,
        t1.Transaction_ID,
        t1.Timestamp
),

max_velocity AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY Customer_ID
            ORDER BY transactions_within_60min DESC,
                     (last_in_window - first_in_window)
        ) AS rn
    FROM velocity
)

SELECT
    Customer_ID,
    transactions_within_60min,
    first_in_window,
    last_in_window,
    EXTRACT(
        EPOCH FROM (last_in_window - first_in_window)
    ) / 60 AS burst_duration_minutes
FROM max_velocity
WHERE rn = 1
  AND transactions_within_60min >= 6
ORDER BY burst_duration_minutes;

38. Alert → Investigation → Resolution

SELECT
    COUNT(DISTINCT a.Alert_ID) AS total_alerts,

    COUNT(DISTINCT a.Investigation_ID)
        FILTER (WHERE a.Investigation_ID IS NOT NULL)
        AS investigated_alerts,

    COUNT(DISTINCT i.Investigation_ID)
        AS total_investigations,

    COUNT(DISTINCT i.Investigation_ID)
        FILTER (WHERE i.Resolution_Code = 'Confirmed Fraud')
        AS confirmed_fraud_cases,

    COUNT(DISTINCT i.Investigation_ID)
        FILTER (WHERE i.Resolution_Code = 'False Positive')
        AS false_positive_cases,

    ROUND(
        100.0 *
        COUNT(DISTINCT a.Investigation_ID)
        FILTER (WHERE a.Investigation_ID IS NOT NULL)
        / COUNT(DISTINCT a.Alert_ID),
        2
    ) AS alert_to_investigation_pct,

    ROUND(
        100.0 *
        COUNT(DISTINCT i.Investigation_ID)
        FILTER (WHERE i.Resolution_Code = 'Confirmed Fraud')
        / NULLIF(COUNT(DISTINCT i.Investigation_ID), 0),
        2
    ) AS investigation_confirmation_pct

FROM fact_alerts a
LEFT JOIN fact_investigation i
    ON a.Investigation_ID = i.Investigation_ID;
